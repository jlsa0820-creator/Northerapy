     1→"use client";
     2→
     3→import { useMemo, useState } from "react";
     4→import { useCrmStore } from "@/lib/store";
     5→import {
     6→  PAQUETE_PRECIO,
     7→  PRECIO_DICLOFENACO,
     8→  TARIFAS,
     9→  EstadoLinea,
    10→  METODOS_PAGO,
    11→  TarifaMode,
    12→} from "@/lib/types";
    13→import {
    14→  calcularEstado,
    15→  deudaDePaciente,
    16→  diasLaboralesSinAsistir,
    17→  esFechaFutura,
    18→  esFinDeSemana,
    19→  fmtFecha,
    20→  hoyISO,
    21→  paqueteInfo,
    22→  plural,
    23→  soles,
    24→} from "@/lib/business";
    25→import { StatusBadge } from "./badges";
    26→import { useToast } from "@/hooks/use-toast";
    27→import { Button } from "@/components/ui/button";
    28→import { Card, CardContent } from "@/components/ui/card";
    29→import { Input } from "@/components/ui/input";
    30→import { Label } from "@/components/ui/label";
    31→import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
    32→import {
    33→  Dialog,
    34→  DialogContent,
    35→  DialogDescription,
    36→  DialogFooter,
    37→  DialogHeader,
    38→  DialogTitle,
    39→} from "@/components/ui/dialog";
    40→import { CalendarPlus, CheckCircle2, Loader2, Search, Wallet } from "lucide-react";
    41→import { cn } from "@/lib/utils";
    42→
    43→interface Props {
    44→  pacienteInicial: number | null;
    45→  onPacienteUsado: () => void;
    46→}
    47→
    48→export function VisitaView({ pacienteInicial, onPacienteUsado }: Props) {
    49→  const { pacientes, asistencias, ingresos, registrarVisita } = useCrmStore();
    50→  const { toast } = useToast();
    51→
    52→  // Paciente efectivo: el preseleccionado (p. ej. desde «Pacientes») o el elegido localmente
    53→  const [idLocal, setIdLocal] = useState<number | null>(null);
    54→  const pacienteId = pacienteInicial ?? idLocal;
    55→
    56→  // Búsqueda
    57→  const [query, setQuery] = useState("");
    58→  const [menuAbierto, setMenuAbierto] = useState(false);
    59→  const [indiceActivo, setIndiceActivo] = useState(0);
    60→
    61→  // Formulario
    62→  const [fecha, setFecha] = useState(hoyISO());
    63→  const [zonas, setZonas] = useState(1);
    64→  // Tarifa efectiva: la elegida por el usuario o la derivada del paquete activo
    65→  const [tarifaElegida, setTarifaElegida] = useState<TarifaMode | null>(null);
    66→  const [tarifaOtro, setTarifaOtro] = useState("");
    67→  const [dicloCant, setDicloCant] = useState(0);
    68→  const [ortoMonto, setOrtoMonto] = useState("");
    69→  const [venderPaquete, setVenderPaquete] = useState(false);
    70→  const [guardando, setGuardando] = useState(false);
    71→  const [cobrando, setCobrando] = useState(false);
    72→  const [metodoGlobal, setMetodoGlobal] = useState<EstadoLinea>("efectivo");
    73→  const [exito, setExito] = useState<string | null>(null);
    74→
    75→  const paciente = useMemo(
    76→    () => pacientes.find((p) => p.paciente_id === pacienteId) ?? null,
    77→    [pacientes, pacienteId]
    78→  );
    79→  const ingresosPaciente = useMemo(
    80→    () => ingresos.filter((i) => i.paciente_id === pacienteId),
    81→    [ingresos, pacienteId]
    82→  );
    83→  const asistencia = useMemo(
    84→    () => asistencias.find((a) => a.paciente_id === pacienteId) ?? null,
    85→    [asistencias, pacienteId]
    86→  );
    87→  const estado = useMemo(
    88→    () => (paciente ? calcularEstado({ paciente, asistencia, ingresos: ingresosPaciente }) : null),
    89→    [paciente, asistencia, ingresosPaciente]
    90→  );
    91→  const deuda = useMemo(() => deudaDePaciente(ingresosPaciente), [ingresosPaciente]);
    92→  const paquete = useMemo(() => paqueteInfo(ingresosPaciente), [ingresosPaciente]);
    93→  const tarifa: TarifaMode = tarifaElegida ?? (paquete?.activo ? "cubierta" : "suelta");
    94→
    95→  const resultados = useMemo(() => {
    96→    const q = query.trim().toLowerCase();
    97→    if (!q) return [];
    98→    return pacientes
    99→      .filter((p) => p.nombre_completo.toLowerCase().includes(q))
   100→      .sort((a, b) => {
   101→        const ia = a.nombre_completo.toLowerCase().indexOf(q);
   102→        const ib = b.nombre_completo.toLowerCase().indexOf(q);
   103→        if (ia !== ib) return ia - ib;
   104→        return a.nombre_completo.localeCompare(b.nombre_completo, "es");
   105→      })
   106→      .slice(0, 8);
   107→  }, [pacientes, query]);
   108→
   109→  // Índice activo recortado al tamaño de resultados actuales
   110→  const indiceSeguro = Math.min(indiceActivo, Math.max(0, resultados.length - 1));
   111→
   112→  const precioZona = tarifa === "otro" ? Number(tarifaOtro || 0) : TARIFAS[tarifa as keyof typeof TARIFAS];
   113→  const montoTerapia = Number.isFinite(precioZona) ? precioZona * zonas : 0;
   114→  const montoDiclo = dicloCant * PRECIO_DICLOFENACO;
   115→  const montoOrto = Number(ortoMonto || 0) || 0;
   116→  const totalVisita = montoTerapia + montoDiclo + montoOrto;
   117→  const totalConPaquete = totalVisita + (venderPaquete ? PAQUETE_PRECIO : 0);
   118→
   119→  const finDeSemanaWarn = fecha && esFinDeSemana(fecha) && fecha === hoyISO();
   120→  const fechaMala = fecha && esFechaFutura(fecha);
   121→
   122→  function seleccionarPaciente(id: number) {
   123→    setIdLocal(id);
   124→    onPacienteUsado();
   125→    setQuery("");
   126→    setMenuAbierto(false);
   127→    setExito(null);
   128→    setVenderPaquete(false);
   129→    setTarifaElegida(null);
   130→  }
   131→
   132→  function onBuscarTeclado(e: React.KeyboardEvent) {
   133→    if (!menuAbierto || resultados.length === 0) return;
   134→    if (e.key === "ArrowDown") {
   135→      e.preventDefault();
   136→      setIndiceActivo((i) => (i + 1) % resultados.length);
   137→    } else if (e.key === "ArrowUp") {
   138→      e.preventDefault();
   139→      setIndiceActivo((i) => (i - 1 + resultados.length) % resultados.length);
   140→    } else if (e.key === "Enter") {
   141→      e.preventDefault();
   142→      seleccionarPaciente(resultados[indiceSeguro].paciente_id);
   143→    } else if (e.key === "Escape") {
   144→      setMenuAbierto(false);
   145→    }
   146→  }
   147→
   148→  async function guardar(cobrarAhora: boolean) {
   149→    if (!paciente) return;
   150→    if (fechaMala) {
   151→      toast({ title: "Fecha inválida", description: "No puedes registrar visitas con fecha futura.", variant: "destructive" });
   152→      return;
   153→    }
   154→    if (cobrarAhora) {
   155→      setCobrando(true);
   156→      return;
   157→    }
   158→    await persistir({
   159→      terapia: { estado: montoTerapia > 0 ? "debe" : "no_aplica", monto: montoTerapia },
   160→      diclofenaco: { estado: montoDiclo > 0 ? "debe" : "no_aplica", monto: montoDiclo },
   161→      ortopedico: { estado: montoOrto > 0 ? "debe" : "no_aplica", monto: montoOrto },
   162→      paquete: venderPaquete ? "debe" : null,
   163→    });
   164→  }
   165→
   166→  async function confirmarCobro() {
   167→    // El método elegido se aplica a todas las líneas de la visita
   168→    await persistir({
   169→      terapia: {
   170→        estado: montoTerapia > 0 ? metodoGlobal : "no_aplica",
   171→        monto: montoTerapia,
   172→      },
   173→      diclofenaco: {
   174→        estado: montoDiclo > 0 ? metodoGlobal : "no_aplica",
   175→        monto: montoDiclo,
   176→      },
   177→      ortopedico: {
   178→        estado: montoOrto > 0 ? metodoGlobal : "no_aplica",
   179→        monto: montoOrto,
   180→      },
   181→      paquete: venderPaquete ? metodoGlobal : null,
   182→    });
   183→    setCobrando(false);
   184→  }
   185→
   186→  async function persistir(m: {
   187→    terapia: { estado: EstadoLinea; monto: number };
   188→    diclofenaco: { estado: EstadoLinea; monto: number };
   189→    ortopedico: { estado: EstadoLinea; monto: number };
   190→    paquete: EstadoLinea | null;
   191→  }) {
   192→    if (!paciente) return;
   193→    setGuardando(true);
   194→    const res = await registrarVisita({
   195→      pacienteId: paciente.paciente_id,
   196→      fecha,
   197→      zonas,
   198→      terapia: m.terapia,
   199→      diclofenaco: m.diclofenaco,
   200→      ortopedico: m.ortopedico,
   201→      venderPaquete: m.paquete ? { estado: m.paquete } : null,
   202→    });
   203→    setGuardando(false);
   204→    if (res.ok) {
   205→      const texto = m.paquete === null && !m.terapia.monto && !m.diclofenaco.monto && !m.ortopedico.monto
   206→        ? "Visita registrada (sin cargos)"
   207→        : m.terapia.estado === "debe" && (m.terapia.monto > 0 || m.diclofenaco.monto > 0 || m.ortopedico.monto > 0)
   208→          ? "Visita registrada. El monto quedó como deuda (estado «debe»)."
   209→          : "Visita y cobro registrados ✓";
   210→      setExito(texto);
   211→      toast({ title: "Guardado en Supabase ✓", description: texto });
   212→      // Reset de la visita (mantiene el paciente para agilizar)
   213→      setDicloCant(0);
   214→      setOrtoMonto("");
   215→      setVenderPaquete(false);
   216→      setTarifaOtro("");
   217→    } else {
   218→      toast({ title: "No se pudo guardar", description: res.error, variant: "destructive" });
   219→    }
   220→  }
   221→
   222→  return (
   223→    <div className="mx-auto max-w-3xl space-y-4">
   224→      <div className="flex items-center gap-3">
   225→        <div className="flex size-10 items-center justify-center rounded-xl border border-teal-500/30 bg-teal-500/10">
   226→          <CalendarPlus className="size-5 text-teal-300" aria-hidden />
   227→        </div>
   228→        <div>
   229→          <h1 className="text-xl font-bold tracking-tight sm:text-2xl">Anotar visita y cobro</h1>
   230→          <p className="text-sm text-muted-foreground">Busca al paciente, anota la terapia y cobra en un solo paso.</p>
   231→        </div>
   232→      </div>
   233→
   234→      {/* Buscador */}
   235→      <div className="relative">
   236→        <div className="relative">
   237→          <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   238→          <Input
   239→            placeholder="Escribe el nombre del paciente…"
   240→            className="h-11 pl-9 text-base"
   241→            value={query}
   242→            onChange={(e) => {
   243→              setQuery(e.target.value);
   244→              setMenuAbierto(true);
   245→            }}
   246→            onFocus={() => setMenuAbierto(true)}
   247→            onKeyDown={onBuscarTeclado}
   248→            aria-label="Buscar paciente por nombre"
   249→            role="combobox"
   250→            aria-expanded={menuAbierto && resultados.length > 0}
   251→          />
   252→        </div>
   253→        {menuAbierto && query.trim() && (
   254→          <ul
   255→            role="listbox"
   256→            className="absolute inset-x-0 top-full z-30 mt-1.5 max-h-80 overflow-y-auto rounded-xl border border-border bg-popover shadow-2xl"
   257→          >
   258→            {resultados.length === 0 && (
   259→              <li className="px-4 py-3 text-sm text-muted-foreground">Sin coincidencias para «{query}».</li>
   260→            )}
   261→            {resultados.map((p, i) => {
   262→              const idx = p.nombre_completo.toLowerCase().indexOf(query.trim().toLowerCase());
   263→              const nombre = idx >= 0 ? (
   264→                <>
   265→                  {p.nombre_completo.slice(0, idx)}
   266→                  <mark className="rounded bg-teal-500/25 px-0.5 text-teal-200">{p.nombre_completo.slice(idx, idx + query.trim().length)}</mark>
   267→                  {p.nombre_completo.slice(idx + query.trim().length)}
   268→                </>
   269→              ) : p.nombre_completo;
   270→              return (
   271→                <li key={p.paciente_id}>
   272→                  <button
   273→                    role="option"
   274→                    aria-selected={i === indiceSeguro}
   275→                    className={cn(
   276→                      "flex w-full items-center justify-between gap-2 px-4 py-2.5 text-left text-sm transition-colors",
   277→                      i === indiceSeguro ? "bg-accent text-accent-foreground" : "hover:bg-secondary/60"
   278→                    )}
   279→                    onMouseEnter={() => setIndiceActivo(i)}
   280→                    onClick={() => seleccionarPaciente(p.paciente_id)}
   281→                  >
   282→                    <span>{nombre}</span>
   283→                    <span className="text-[11px] text-muted-foreground">ID {p.paciente_id}</span>
   284→                  </button>
   285→                </li>
   286→              );
   287→            })}
   288→          </ul>
   289→        )}
   290→      </div>
   291→
   292→      {/* Ficha del paciente */}
   293→      {paciente && estado && (
   294→        <Card className="border-border/70 bg-card/80">
   295→          <CardContent className="p-4 sm:p-5">
   296→            <div className="flex flex-wrap items-start justify-between gap-3">
   297→              <div>
   298→                <p className="text-base font-bold">{paciente.nombre_completo}</p>
   299→                <p className="text-xs text-muted-foreground">
   300→                  {paciente.diagnostico ?? "Sin diagnóstico registrado"}
   301→                </p>
   302→              </div>
   303→              <StatusBadge estado={estado} />
   304→            </div>
   305→            <div className="mt-3 grid grid-cols-2 gap-3 text-sm sm:grid-cols-4">
   306→              <div>
   307→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Sesiones</p>
   308→                <p className="num-tabular mt-0.5 font-semibold">
   309→                  {paciente.sesiones_totales}
   310→                  {paciente.sesiones_programadas ? ` / ${paciente.sesiones_programadas}` : ""}
   311→                </p>
   312→              </div>
   313→              <div>
   314→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Última vez</p>
   315→                <p className="mt-0.5">
   316→                  {asistencia?.ultima_asistencia ? (
   317→                    <>
   318→                      {fmtFecha(asistencia.ultima_asistencia)}
   319→                      <span className="block text-[11px] text-muted-foreground">
   320→                        hace {plural(diasLaboralesSinAsistir(asistencia.ultima_asistencia) ?? 0, "día laboral", "días laborales")}
   321→                      </span>
   322→                    </>
   323→                  ) : (
   324→                    <span className="text-muted-foreground">Primera visita</span>
   325→                  )}
   326→                </p>
   327→              </div>
   328→              <div>
   329→                <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Deuda</p>
   330→                <p className={cn("num-tabular mt-0.5 font-semibold", deuda > 0 ? "text-rose-400" : "text-emerald-400")}>
   331→                  {deuda > 0 ? soles(deuda) : "Al día"}
   332→                </p>
   333→              </div>
   334→              {paquete && (
   335→                <div>
   336→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground">Paquete</p>
   337→                  <p className={cn("mt-0.5 font-semibold", paquete.activo ? "text-teal-300" : "text-muted-foreground")}>
   338→                    {paquete.activo ? `Activo · quedan ${paquete.restantes}` : "Agotado"}
   339→                  </p>
   340→                </div>
   341→              )}
   342→            </div>
   343→
   344→            {paquete?.activo && (
   345→              <p className="mt-3 rounded-lg border border-teal-500/25 bg-teal-500/8 px-3 py-2 text-xs text-teal-200">
   346→                Paquete de 5 sesiones activo: le quedan {paquete.restantes} (usadas {paquete.usadas} de 5, comprado el {fmtFecha(paquete.fechaCompra)}). La tarifa «Cubierta por paquete» quedó seleccionada.
   347→              </p>
   348→            )}
   349→          </CardContent>
   350→        </Card>
   351→      )}
   352→
   353→      {paciente && (
   354→        <Card className="border-border/70 bg-card/80">
   355→          <CardContent className="space-y-5 p-4 sm:p-5">
   356→            {exito && (
   357→              <p className="flex items-center gap-2 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-3 py-2.5 text-sm font-medium text-emerald-300">
   358→                <CheckCircle2 className="size-4 shrink-0" aria-hidden /> {exito}
   359→              </p>
   360→            )}
   361→
   362→            {/* Fecha */}
   363→            <div className="grid gap-4 sm:grid-cols-2">
   364→              <div className="space-y-1.5">
   365→                <Label htmlFor="fecha-visita">Fecha de la visita *</Label>
   366→                <Input
   367→                  id="fecha-visita"
   368→                  type="date"
   369→                  max={hoyISO()}
   370→                  value={fecha}
   371→                  onChange={(e) => setFecha(e.target.value)}
   372→                  className={cn("w-full sm:w-44", fechaMala && "border-rose-500/60")}
   373→                />
   374→                {fechaMala && <p className="text-xs text-rose-400">No se permiten fechas futuras.</p>}
   375→                {finDeSemanaWarn && <p className="text-xs text-amber-400">Es fin de semana: confirma que la visita fue hoy.</p>}
   376→              </div>
   377→              <div className="space-y-1.5">
   378→                <Label>Zonas a tratar *</Label>
   379→                <RadioGroup value={String(zonas)} onValueChange={(v) => setZonas(Number(v))} className="flex gap-2">
   380→                  {[1, 2].map((z) => (
   381→                    <label
   382→                      key={z}
   383→                      className={cn(
   384→                        "flex min-h-11 flex-1 cursor-pointer items-center justify-center gap-2 rounded-lg border text-sm font-medium transition-colors",
   385→                        zonas === z ? "border-teal-500/50 bg-teal-500/10 text-teal-200" : "border-border/70 text-muted-foreground hover:border-border"
   386→                      )}
   387→                    >
   388→                      <RadioGroupItem value={String(z)} id={`zona-${z}`} className="sr-only" />
   389→                      {plural(z, "zona", "zonas")}
   390→                    </label>
   391→                  ))}
   392→                </RadioGroup>
   393→              </div>
   394→            </div>
   395→
   396→            {/* Tarifa */}
   397→            <div className="space-y-1.5">
   398→              <Label>Tarifa de la terapia</Label>
   399→              <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   400→                {(
   401→                  [
   402→                    { id: "cubierta", titulo: "Cubierta", sub: "S/ 0 · incluida en paquete" },
   403→                    { id: "paquete", titulo: "Paquete", sub: `S/ ${TARIFAS.paquete} por zona` },
   404→                    { id: "suelta", titulo: "Suelta", sub: `S/ ${TARIFAS.suelta} por zona` },
   405→                  ] as { id: TarifaMode; titulo: string; sub: string }[]
   406→                ).map((op) => (
   407→                  <button
   408→                    key={op.id}
   409→                    type="button"
   410→                    onClick={() => setTarifaElegida(op.id)}
   411→                    className={cn(
   412→                      "rounded-lg border px-3 py-2.5 text-left transition-colors",
   413→                      tarifa === op.id ? "border-teal-500/50 bg-teal-500/10" : "border-border/70 hover:border-border"
   414→                    )}
   415→                  >
   416→                    <span className={cn("block text-sm font-semibold", tarifa === op.id ? "text-teal-200" : "")}>{op.titulo}</span>
   417→                    <span className="block text-[11px] text-muted-foreground">{op.sub}</span>
   418→                  </button>
   419→                ))}
   420→                <div className={cn("rounded-lg border px-3 py-2 transition-colors", tarifa === "otro" ? "border-teal-500/50 bg-teal-500/10" : "border-border/70")}>
   421→                  <button type="button" className="block w-full text-left text-sm font-semibold" onClick={() => setTarifaElegida("otro")}>
   422→                    Otro monto
   423→                  </button>
   424→                  <Input
   425→                    aria-label="Monto personalizado por zona"
   426→                    type="number"
   427→                    min={0}
   428→                    step="0.5"
   429→                    placeholder="S/ libre"
   430→                    className="mt-1 h-7 border-border/70 text-xs"
   431→                    value={tarifaOtro}
   432→                    onChange={(e) => {
   433→                      setTarifaElegida("otro");
   434→                      setTarifaOtro(e.target.value);
   435→                    }}
   436→                  />
   437→                </div>
   438→              </div>
   439→              {tarifa === "cubierta" && !paquete?.activo && (
   440→                <p className="text-xs text-amber-400">
   441→                  Este paciente no tiene paquete activo: la sesión quedará en S/ 0. Revisa que sea correcto.
   442→                </p>
   443→              )}
   444→            </div>
   445→
   446→            {/* Extras */}
   447→            <div className="grid gap-4 sm:grid-cols-2">
   448→              <div className="space-y-1.5">
   449→                <Label htmlFor="diclo">Diclofenaco (S/ {PRECIO_DICLOFENACO} c/u)</Label>
   450→                <div className="flex items-center gap-2">
   451→                  <Button type="button" variant="outline" size="icon" className="size-9" onClick={() => setDicloCant(Math.max(0, dicloCant - 1))} aria-label="Quitar diclofenaco">−</Button>
   452→                  <Input
   453→                    id="diclo"
   454→                    type="number"
   455→                    min={0}
   456→                    className="num-tabular w-20 text-center"
   457→                    value={dicloCant}
   458→                    onChange={(e) => setDicloCant(Math.max(0, Number(e.target.value || 0)))}
   459→                  />
   460→                  <Button type="button" variant="outline" size="icon" className="size-9" onClick={() => setDicloCant(dicloCant + 1)} aria-label="Agregar diclofenaco">+</Button>
   461→                  <span className="num-tabular text-sm text-muted-foreground">= {soles(montoDiclo)}</span>
   462→                </div>
   463→              </div>
   464→              <div className="space-y-1.5">
   465→                <Label htmlFor="orto">Ortopédico (monto libre)</Label>
   466→                <Input
   467→                  id="orto"
   468→                  type="number"
   469→                  min={0}
   470→                  step="0.5"
   471→                  placeholder="S/ 0.00"
   472→                  className="num-tabular w-40"
   473→                  value={ortoMonto}
   474→                  onChange={(e) => setOrtoMonto(e.target.value)}
   475→                />
   476→              </div>
   477→            </div>
   478→
   479→            {/* Vender paquete */}
   480→            <div
   481→              className={cn(
   482→                "flex items-center justify-between gap-3 rounded-lg border px-3.5 py-3 transition-colors",
   483→                venderPaquete ? "border-teal-500/40 bg-teal-500/8" : "border-border/70 bg-secondary/30"
   484→              )}
   485→            >
   486→              <div>
   487→                <p className="text-sm font-semibold">¿Le vendes el paquete de 5 sesiones?</p>
   488→                <p className="text-[11px] text-muted-foreground">
   489→                  Cargo único de {soles(PAQUETE_PRECIO)} fechado hoy: la sesión de este día ya cuenta y quedan 4.
   490→                  {paquete?.activo ? " (Ya tiene un paquete activo — revisa antes de vender otro.)" : ""}
   491→                </p>
   492→              </div>
   493→              <button
   494→                type="button"
   495→                role="switch"
   496→                aria-checked={venderPaquete}
   497→                onClick={() => setVenderPaquete(!venderPaquete)}
   498→                className={cn(
   499→                  "relative h-7 w-12 shrink-0 rounded-full transition-colors",
   500→                  venderPaquete ? "bg-teal-500" : "bg-secondary border border-border"
   501→                )}
   502→              >
   503→                <span
   504→                  className={cn(
   505→                    "absolute top-1 size-5 rounded-full bg-white transition-all",
   506→                    venderPaquete ? "left-6" : "left-1"
   507→                  )}
   508→                />
   509→              </button>
   510→            </div>
   511→
   512→            {/* Desglose */}
   513→            <div className="rounded-xl border border-border/70 bg-background/50 p-4">
   514→              <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">Desglose</p>
   515→              <ul className="num-tabular space-y-1 text-sm">
   516→                <li className="flex justify-between"><span className="text-muted-foreground">Terapia ({tarifa === "otro" ? "monto libre" : tarifa} × {zonas} {zonas === 1 ? "zona" : "zonas"})</span><span>{soles(montoTerapia)}</span></li>
   517→                {dicloCant > 0 && (
   518→                  <li className="flex justify-between"><span className="text-muted-foreground">Diclofenaco × {dicloCant}</span><span>{soles(montoDiclo)}</span></li>
   519→                )}
   520→                {montoOrto > 0 && (
   521→                  <li className="flex justify-between"><span className="text-muted-foreground">Ortopédico</span><span>{soles(montoOrto)}</span></li>
   522→                )}
   523→                {venderPaquete && (
   524→                  <li className="flex justify-between"><span className="text-muted-foreground">Paquete de 5 sesiones</span><span>{soles(PAQUETE_PRECIO)}</span></li>
   525→                )}
   526→                <li className="flex justify-between border-t border-border/60 pt-2 text-base font-bold">
   527→                  <span>Total {venderPaquete ? "(incluye paquete)" : ""}</span>
   528→                  <span className="text-teal-300">{soles(totalConPaquete)}</span>
   529→                </li>
   530→              </ul>
   531→            </div>
   532→
   533→            {/* Acciones */}
   534→            <div className="flex flex-col gap-2 sm:flex-row">
   535→              <Button
   536→                variant="outline"
   537→                className="flex-1 border-amber-500/40 text-amber-300 hover:bg-amber-500/10 hover:text-amber-200"
   538→                onClick={() => void guardar(false)}
   539→                disabled={guardando || !!fechaMala}
   540→              >
   541→                {guardando ? <Loader2 className="size-4 animate-spin" aria-hidden /> : "Guardar visita (cobrar después)"}
   542→              </Button>
   543→              <Button className="flex-1 font-semibold" onClick={() => void guardar(true)} disabled={guardando || !!fechaMala}>
   544→                <Wallet className="size-4" aria-hidden /> Guardar y cobrar ahora
   545→              </Button>
   546→            </div>
   547→            <p className="text-center text-[11px] text-muted-foreground">
   548→              Se creará el ingreso en Supabase y se actualizarán sus sesiones y resumen de asistencia.
   549→            </p>
   550→          </CardContent>
   551→        </Card>
   552→      )}
   553→
   554→      {!paciente && !query && (
   555→        <Card className="border-dashed border-border/70 bg-card/40">
   556→          <CardContent className="flex flex-col items-center gap-2 py-10 text-center">
   557→            <Search className="size-8 text-muted-foreground/50" aria-hidden />
   558→            <p className="text-sm text-muted-foreground">
   559→              Empieza escribiendo el nombre del paciente para ver su ficha.
   560→            </p>
   561→          </CardContent>
   562→        </Card>
   563→      )}
   564→
   565→      {/* Modal de cobro */}
   566→      <Dialog open={cobrando} onOpenChange={setCobrando}>
   567→        <DialogContent className="border-border/70 bg-card">
   568→          <DialogHeader>
   569→            <DialogTitle>Cobrar {soles(totalConPaquete)}</DialogTitle>
   570→            <DialogDescription>
   571→              Elige el método de pago: se aplicará a todas las líneas de esta visita.
   572→            </DialogDescription>
   573→          </DialogHeader>
   574→
   575→          <div className="space-y-4">
   576→            <div>
   577→              <p className="mb-1.5 text-xs font-semibold uppercase tracking-wider text-muted-foreground">Método de pago</p>
   578→              <div className="grid grid-cols-3 gap-2">
   579→                {METODOS_PAGO.map((m) => (
   580→                  <button
   581→                    key={m}
   582→                    type="button"
   583→                    onClick={() => setMetodoGlobal(m)}
   584→                    className={cn(
   585→                      "min-h-11 rounded-lg border text-sm font-medium capitalize transition-colors",
   586→                      metodoGlobal === m ? "border-teal-500/50 bg-teal-500/10 text-teal-200" : "border-border/70 hover:border-border"
   587→                    )}
   588→                  >
   589→                    {m === "efectivo" ? "Efectivo" : m === "yape" ? "Yape" : "Tarjeta"}
   590→                  </button>
   591→                ))}
   592→              </div>
   593→            </div>
   594→
   595→            <div className="space-y-2 rounded-lg border border-border/60 bg-background/40 p-3">
   596→              {(montoTerapia > 0 || (venderPaquete && PAQUETE_PRECIO > 0)) && (
   597→                <div className="flex items-center justify-between gap-2 text-sm">
   598→                  <span className="text-muted-foreground">Terapia{venderPaquete ? " + paquete" : ""}</span>
   599→                  <span className="num-tabular font-semibold">{soles(montoTerapia + (venderPaquete ? PAQUETE_PRECIO : 0))}</span>
   600→                </div>
   601→              )}
   602→              {(montoDiclo > 0 || montoOrto > 0) && (
   603→                <div className="flex items-center justify-between gap-2 text-sm">
   604→                  <span className="text-muted-foreground">Diclofenaco{montoOrto > 0 ? " + ortopédico" : ""}</span>
   605→                  <span className="num-tabular font-semibold">{soles(montoDiclo + montoOrto)}</span>
   606→                </div>
   607→              )}
   608→              <div className="flex justify-between border-t border-border/60 pt-2 text-sm font-bold">
   609→                <span>Total a cobrar</span>
   610→                <span className="num-tabular text-teal-300">{soles(totalConPaquete)}</span>
   611→              </div>
   612→            </div>
   613→          </div>
   614→
   615→          <DialogFooter>
   616→            <Button variant="ghost" onClick={() => setCobrando(false)} disabled={guardando}>Cancelar</Button>
   617→            <Button onClick={() => void confirmarCobro()} disabled={guardando}>
   618→              {guardando ? <Loader2 className="size-4 animate-spin" aria-hidden /> : `Confirmar cobro de ${soles(totalConPaquete)}`}
   619→            </Button>
   620→          </DialogFooter>
   621→        </DialogContent>
   622→      </Dialog>
   623→    </div>
   624→  );
   625→}
